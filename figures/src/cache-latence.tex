% Latence moyenne d'une lecture selon le taux de succès du cache Redis
% (t = 61 µs pour le cache ; base : 70 µs pour une clé primaire, 127 ms pour
% le top 10 d'une catégorie, mesurés par outils/experiences/ch16/niveaux.py).
\documentclass[border=6pt]{standalone}
\usepackage{sd}
\begin{document}
\begin{tikzpicture}[papier, x=1.1mm, y=11mm, every node/.append style={font=\sffamily\small}, note/.append style={font=\sffamily\small}]
  % axe y logarithmique : y = log10(µs), calculé comme 3 + log10(ms) (limite de TeX)
  \draw[ligne] (0,1.3) -- (102,1.3);
  \draw[ligne] (0,1.3) -- (0,5.4);
  \foreach \v/\l in {2/{100 µs}, 3/{1 ms}, 4/{10 ms}, 5/{100 ms}} {\draw[ligne=Dcloud] (0,\v) -- (-1,\v); \node[note, left] at (-1,\v) {\l};}
  \foreach \h in {0,20,...,100} {\draw[ligne] (\h,1.3) -- (\h,1.15); \node[note, below] at (\h,1.15) {\h\ \%};}
  \node[axe, anchor=north] at (51,0.65) {taux de succès du cache};
  % top 10 : t = 61 + (1 - h) * 127000 µs
  \draw[Dred, line width=1.1pt, domain=0:99.9, samples=80] plot (\x, {3 + log10(0.061 + (1 - \x/100) * 126.584)});
  \draw[Dred, dash pattern=on 3pt off 2pt] (0,{3 + log10(126.584)}) -- (100,{3 + log10(126.584)});
  \node[et=Dred, anchor=south east] at (100,{3 + log10(126.584)}) {top 10 sans cache : 127 ms};
  \node[et=Dred, anchor=north west] at (20,4.7) {top 10 avec cache};
  % clé primaire : t = 61 + (1 - h) * 70 µs
  \draw[Dblue, line width=1.1pt, domain=0:100, samples=40] plot (\x, {3 + log10(0.061 + (1 - \x/100) * 0.070)});
  \draw[Dblue, dash pattern=on 3pt off 2pt] (0,{log10(70)}) -- (100,{log10(70)});
  \node[et=Dblue, anchor=north west] at (1,{log10(70)-0.04}) {clé primaire sans cache : 70 µs};
  \node[et=Dblue, anchor=south west] at (1,{log10(131)+0.03}) {clé primaire avec cache : plus lent sous 87 \% de succès};
  \fill[Dblue] (87,{log10(70)}) circle[radius=1.2pt];
\end{tikzpicture}
\end{document}

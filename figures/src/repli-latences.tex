\begin{tikzpicture}[x=1mm, y=1mm, note/.append style={font=\sffamily\small},
    barre/.style={draw=#1, fill=#1Bg, line width=0.6pt}]
  \def\H#1{#1*0.8}
  \draw[ligne] (0,0) -- (150,0);
  \draw[ligne] (0,0) -- (0,52);
  \foreach \v in {0, 20, 40, 60} {
    \draw[ligne] (0,{\H{\v}}) -- (-1.5,{\H{\v}});

  \node[note, rotate=90, anchor=south] at (-10,24) {latence médiane (ms)};
  % groupes : aller-retour 0, 1 et 50 ms ; modes : async, off, local, remote_write, on, remote_apply

  \foreach \i/\t in {0/asynchrone, 1/off, 2/local, 3/remote\_write, 4/on, 5/remote\_apply} {

  \node[note, anchor=west, align=left] at (20,40) {sans attendre la réplique :\\0,1 ms (off) ou ≈ 5 ms};

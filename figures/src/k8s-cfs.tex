% Une requête de 50 ms de calcul sous trois limites de processeur : le quota CFS
% est remis à zéro toutes les 100 ms ; au-delà, le conteneur attend la période
% suivante. Latences médianes mesurées par outils/experiences/ch51/ressources.py.
\documentclass[border=6pt]{standalone}
\usepackage{sd}
\begin{document}
\begin{tikzpicture}[x=0.42mm, y=1mm, note/.append style={font=\sffamily\small},
    lab/.style={font=\sffamily\small, text=#1, align=left}, lab/.default=Dink]
  \def\calcul#1#2#3{\fill[Dteal] (#1,#3-3) rectangle (#2,#3+3);}
  \def\bride#1#2#3{\fill[DredBg] (#1,#3-3) rectangle (#2,#3+3); \draw[Dred, line width=0.4pt] (#1,#3-3) rectangle (#2,#3+3);}
  % axe et périodes
  \draw[ligne] (0,-4) -- (310,-4);
  \foreach \t in {0,100,200,300} {\draw[Dcloud, line width=0.4pt, dash pattern=on 1.5pt off 1.5pt] (\t,-4) -- (\t,40);
    \node[note, below] at (\t,-4) {\t\,ms};}
  \node[axe, anchor=north] at (150,-10) {temps depuis l’arrivée de la requête ; une période CFS toutes les 100 ms};
  % 1 cœur
  \node[lab, anchor=east] at (-4,32) {limite 1 cœur\\quota 100 ms};
  \calcul{0}{50}{32}
  \node[lab=Dteal, anchor=west] at (54,32) {fini à 50 ms ; mesuré 51 ms};
  % 500m
  \node[lab, anchor=east] at (-4,18) {limite 500m\\quota 50 ms};
  \calcul{0}{50}{18} \bride{50}{100}{18} \calcul{100}{101}{18}
  \node[lab=Dteal, anchor=west] at (106,18) {avec le traitement HTTP, la requête dépasse 50 ms :\\la fin attend la période suivante ; mesuré 101 ms};
  % 200m
  \node[lab, anchor=east] at (-4,4) {limite 200m\\quota 20 ms};
  \calcul{0}{20}{4} \bride{20}{100}{4} \calcul{100}{120}{4} \bride{120}{200}{4} \calcul{200}{210}{4}
  \node[lab=Dteal, anchor=west] at (214,4) {au moins 210 ms ;\\mesuré 282 à 290 ms};
  % légende
  \fill[Dteal] (0,-24) rectangle (8,-20); \node[lab, anchor=west] at (10,-22) {calcul};
  \fill[DredBg] (50,-24) rectangle (58,-20); \draw[Dred, line width=0.4pt] (50,-24) rectangle (58,-20);
  \node[lab, anchor=west] at (60,-22) {quota épuisé : le conteneur est bridé jusqu’à la période suivante};
\end{tikzpicture}
\end{document}

\begin{tikzpicture}[x=3.4mm, note/.append style={font=\sffamily\small},
    courbe/.style={draw=#1, line width=1.2pt, line cap=round}]
  \newcommand{\trace}[3][1]{% #1 : échelle verticale (mm par unité), #2 : données, #3 : couleur
    \foreach \x/\y [remember=\x as \xa (initially -1), remember=\y as \ya (initially 0)] in #2 {
      \ifnum\xa>-1 \draw[courbe=#3] (\xa+0.5,{\ya*#1}) -- (\x+0.5,{\y*#1}); \fi
    }}
  \newcommand{\cadre}[3]{% #1 : hauteur (mm), #2 : graduations "valeur/position", #3 : titre
    \fill[DslateZ] (10,0) rectangle (20,#1);
    \draw[ligne] (0,0) -- (40.3,0);
    \draw[ligne] (0,0) -- (0,#1);
    \foreach \v/\p in {#2} {\draw[ligne] (0,\p) -- (-0.3,\p); \node[note, anchor=east] at (-0.3,\p) {\v};}
    \node[axe, anchor=south west] at (0,#1+1) {#3};}
  % tentatives reçues : 0,025 mm par requête
  \begin{scope}[y=1mm]
    \cadre{50}{0/0, 500/12.5, 1000/25, 1500/37.5, 2000/50}{tentatives reçues par le service, par seconde}
    \draw[Dred, line width=0.7pt, dash pattern=on 3pt off 2pt] (0,12.5) -- (10,12.5) -- (10,4.17) -- (20,4.17) -- (20,12.5) -- (40,12.5);
    \node[note, text=Dred, anchor=south west] at (24,13) {capacité : 500/s (167/s pendant le ralentissement)};
    \trace[0.025]{\recuAucun}{Dslate}
    \trace[0.025]{\recuBudget}{Damber}
    \trace[0.025]{\recuPrevue}{Dteal}
    \trace[0.025]{\recuImmediat}{Dred}
  \end{scope}
  % réussites
  \begin{scope}[y=1mm, yshift=-58mm]
    \cadre{40}{0/0, 100/9, 200/18, 300/27, 400/36}{requêtes réussies, par seconde}
    \node[note, align=center] at (15,33) {service 3 fois\\plus lent};
    \trace[0.09]{\okAucun}{Dslate}
    \trace[0.09]{\okBudget}{Damber}
    \trace[0.09]{\okPrevue}{Dteal}
    \trace[0.09]{\okImmediat}{Dred}
    \foreach \t in {0,10,20,30,40} {\draw[ligne] (\t,0) -- (\t,-1); \node[note, below] at (\t,-1) {\t};}
    \node[axe, anchor=north] at (20,-6) {temps (s)};
  \end{scope}
  % légende
  \begin{scope}[shift={(42,-20mm)}]
    \foreach \c/\y/\t in {Dslate/0/{une seule tentative}, Dred/-8/{3 nouvelles tentatives\\immédiates (de même avec\\recul et gigue, ou\\échéance passée)}, Damber/-28/{budget de 10 \%}, Dteal/-36/{3 nouvelles tentatives ;\\le service écarte ce qui\\finirait après l'échéance}} {
      \draw[courbe=\c] (0,\y mm) -- (1.5,\y mm);
      \node[note, anchor=north west, align=left] at (1.8,\y mm+2mm) {\t};
    }
  \end{scope}
\end{tikzpicture}
